    Реализовать функцию, которая принимает массив значений и колбэк по результату которого будут группироваться значения¶

    // Пример
const arr = [6.1, 6.3, 4.2];
const callbackFunc = Math.floor;

const result = {
    '4': [4.2],
    '6': [6.1, 6.3]
}
    Ответ 
    const groupBy = (arr: number[], callbackFunc: (value: number) => number) => {
    const map = new Map();

    arr.forEach((value: number) => {
        const callbackResult: number = callbackFunc(value);
        const isAlreadyExistedinMap = map.has(callbackResult);
        isAlreadyExistedinMap ? map.get(callbackResult).push(value) : map.set(callbackResult, [value]);
    });

    return Object.fromEntries(map);
}

    Найти сумму всех элементов небинарного дерева. Использовать рекурсивный метод и итеративный метод¶

    const tree = {
  v: 5,
  c: [
    {
      v: 10,
      c: [
        {
          v: 11
        }
      ],
    },
    {
      v: 7,
      c: [
        {
          v: 5,
          c: [
            {
              v: 1
            }
          ]
        }
      ]
    },
    {
      v: 5,
      c: [
        {
          v: 10,
        },
        {
          v: 15
        }
      ]
    }
  ]
}
    Ответ 
    const recursive = (tree) => {
  if (!tree.hasOwnProperty('c')) { return tree.v; }

  let sum = tree.v;
  tree.c.forEach(node => {
    sum += recursive(node);
  })

  return sum;
}

const iteration = (tree) => {
  if (!tree.c.length) { return tree.v || 0; }

  const stack = [];
  let sum = tree.v;

  tree.c?.forEach(node => {
    stack.push(node);
  });

  while (stack.length) {
    const node = stack.pop();
    sum += node.v;
    node['c']?.forEach(n => {
      stack.push(n);
    })
  }

  return sum;
}

  return str.split(search).join(replace);

// or

function replaceSubstring(str, search, replace) {  
  let result = '';  

  for (let i = 0; i < str.length; i++) {  
    if (str.substring(i, i + search.length) === search) {  
      result += replace;  
      i += search.length - 1;  
    } else {  
      result += str[i];  
    }  
  }  

  return result;  
}

    Проверить строку на полиндром¶

    Ответ 
    function isPalindrome(str) {
  const cleaned = removeCommasAndSpaces(str).toLowerCase();
  for (let i = 0, j = cleaned.length - 1; i !== j; i++, j--) {
    if (cleaned[i] !== cleaned[j]) {
      return false;
    }
  }

  return true;
}

function removeCommasAndSpaces(str) {
  return str.split(',').join('').split(' ').join('');
}

console.log(isPalindrome('A man, a plan, a canal, Panama')) // true
console.log(isPalindrome('hello world')) // true

    Array.prototype.myFilter = function(callbackFn) {
  const result = [];
  for (let i = 0; i < this.length; i++) {
    if (callbackFn(this[i], i, this)) {
      result.push(this[i]);
    }
  }

  return result;
}

    Восстановить маршрут по билетам¶

    console.log(reconstructTrip([
  {from: 'Спб', to: 'Минск'},
  {from: 'Киев', to: 'Новосибриск'},
  {from: 'Череповец', to: 'Москва'},
  {from: 'Минск', to: 'Киев'},
  {from: 'Москва', to: 'Спб'},
]));
    Ответ 
    function reconstructTrip(tickets) {
  const map = new Map();

  for (let ticket of tickets) {
    map.set(ticket.from, ticket.to);
  }

  let start = [...map.keys()].find(from => ![...map.values()].includes(from));

  const routes = [];
  for (let i = 0; i < tickets.length; i++) {
    if (!start) { break; }
    routes.push({from: start, to: map.get(start)});
    start = map.get(start);
  }

  return routes;
}

    function promisify(fn) {
  return function (...args) {
    console.log(...args);
    return new Promise((resolve, reject) => {
      fn(...args, (err, result) => {
        if (err) { return reject(err); }
        resolve(result);
      })
    })
  }
}

    Объединение интервалов¶

    const input1 = [[1, 3], [2, 6], [8, 10], [15, 18]]; // [[1, 6], [8, 10], [15, 18]]  
const input2 = [[1, 3], [2, 6], [5, 10], [15, 18]]; // [[1, 10], [15, 18]]  
const input3 = [[1, 4], [4, 5]] // [1, 5]  
const input4 = [[1, 15], [4, 10], [20, 25], [27, 28]]; // [[ 1, 15 ], [ 20, 25 ], [ 27, 28 ]]
    Ответ 


    function mergeIntervals(intervals) {  
  const mergedIntervals = [intervals[0]];  

  for (const interval of intervals) {  
    const recent = mergedIntervals[mergedIntervals.length - 1];  
    if (recent[1] >= interval[0]) {  
      recent[1] = Math.max(interval[1], recent[1]);  
    } else {  
      mergedIntervals.push(interval);  
    }  
  }  

  return mergedIntervals;  
}
