## Функция принимает два объекта с различными ключами. Необходимо вернуть первый объект с обновленными ключами из второго объекта (Только те, которые совпадают)
```ts
// Функция принимает два объекта с различными ключами
// Необходимо вернуть первый обхект с обновленными ключами из второго объекта
// Только те, которые совпадают
  
// Пример
const obj1 = {
	foo: 'foo',
	bar: 'bar'
}

const obj2 = {
	bar: 'foo',
	some: 'some'
}

const result = {
	foo: 'foo',
	bar: 'foo'
}
```
> [!faq]- Ответ
> 
> ```ts
> const mergeSomeKyesOfObject = (obj1: UnknownObject, obj2: UnknownObject) => {
> 	const copiedObject: UnknownObject = structuredClone(obj1);
> 	for (let key in copiedObject) {
>         if (key in obj2) {
>             copiedObject[key] = obj2[key]
>         }
>     }
>     return copiedObject;
> }
> ```

## Реализовать функцию, которая принимает массив значений и колбэк по результату которого будут группироваться значения
```ts 
// Пример
const arr = [6.1, 6.3, 4.2];
const callbackFunc = Math.floor;

const result = {
    '4': [4.2],
    '6': [6.1, 6.3]
}
```
> [!faq]- Ответ
> ```ts
> const groupBy = (arr: number[], callbackFunc: (value: number) => number) => {
>     const map = new Map();
>     
>     arr.forEach((value: number) => {
>         const callbackResult: number = callbackFunc(value);
>         const isAlreadyExistedinMap = map.has(callbackResult);
>         isAlreadyExistedinMap ? map.get(callbackResult).push(value) : map.set(callbackResult, [value]);
>     });
> 
>     return Object.fromEntries(map);
> }
> 
> console.log(groupBy(arr, callbackFunc));
> ```

## Написать метод delay для функции, который вызовет заданную функцию спустя какое то количество времени с заданными аргументами
```ts
const START = Date.now();

function someFn() {
    console.log('time', Date.now() - START);
    console.log('args', arguments);
}

const f = someFn.delay(1000);
f('arg1', 'arg2', 1, 2);
```
> [!faq]- Ответ
> 
> ```ts
> Function.prototype.delay = function(timeMs) {
>   return (...args) => { // обратить внимание на стрелочную функцию (сохраняем контекст + параметры функции)
>     setTimeout(() => {
>       this.call(this, ...args); // вызов функции через call, чтобы сохранить ее параметры
>     }, timeMs);
>   }
> }
> ```

## Найти сумму всех элементов небинарного дерева. Использовать рекурсивный метод и итеративный метод
```js
const tree = {
  v: 5,
  c: [
    {
      v: 10,
      c: [
        {
          v: 11
        }
      ],
    },
    {
      v: 7,
      c: [
        {
          v: 5,
          c: [
            {
              v: 1
            }
          ]
        }
      ]
    },
    {
      v: 5,
      c: [
        {
          v: 10,
        },
        {
          v: 15
        }
      ]
    }
  ]
}
```
> [!faq]- Ответ
> 
> ```js
> const recursive = (tree) => {
>   if (!tree.hasOwnProperty('c')) { return tree.v; }
>   
>   let sum = tree.v;
>   tree.c.forEach(node => {
>     sum += recursive(node);
>   })
>   
>   return sum;
> }
> 
> const iteration = (tree) => {
>   if (!tree.c.length) { return tree.v || 0; }
>   
>   const stack = [];
>   let sum = tree.v;
>   
>   tree.c?.forEach(node => {
>     stack.push(node);
>   });
>   
>   while (stack.length) {
>     const node = stack.pop();
>     sum += node.v;
>     node['c']?.forEach(n => {
>       stack.push(n);
>     })
>   }
>   
>   return sum;
> }
> ```
